# Профиль начального обмена «как h2 браузера» после рукопожатия (данные, обновляемая таблица). Парсится src/mimic.rs.
# Все размеры - длина ТЕЛА записи TLS (шифртекст с тегом, как в заголовке записи на проводе); содержимое записей наши - случайное.
# Формат строки: <ключ> <значение(я)>; диапазоны "lo hi" равномерные, целые.
#
# Эталон: Chromium 155.0.8059.84 (хост, --no-proxy-server, headless) -> nginx 1.31 TLS1.3+h2 (X25519MLKEM768), 13 полных h2-сессий
# (8 на "/", 5 на 96 КиБ ресурс "/big"), tools/recsniff.py (границы записей на TCP-прокси); см. RESULTS.md §6.
# Последовательность после Finished клиента (53), c = клиент, s = сервер:
#   c: 87 (preface 24 + SETTINGS 33 + WINDOW_UPDATE 13 + тег), c: 465 (HEADERS GET /; для "/big" 469: +1 Б на символ пути)
#   s: 266, 266 (билеты), 57 (SETTINGS+WINDOW_UPDATE), затем [26 ACK, 118 HEADERS+DATA] либо [104 HEADERS, 40 DATA] (ACK слит)
#   c: 26 (SETTINGS ACK)    -> дальше запросы; ответ на большой ресурс: 146/26 + 16401 x n (+ хвост 173/53)
# Клиент отправляет 87 и 465 сразу после своего Finished, НЕ дожидаясь SETTINGS сервера; ACK (26) - после получения SETTINGS.

# клиент: первая запись пачки (preface+SETTINGS+WINDOW_UPDATE): всегда 87
c_settings 87 87
# клиент: HEADERS запроса: 465 для "/", растёт с длиной пути/кук; берём диапазон типичных GET без кук
c_headers 465 525
# клиент: SETTINGS ACK
c_ack 26 26
# сервер, если снять с сайта не удалось: SETTINGS+WU, ACK, HEADERS+DATA (nginx, маленькая страница)
s_default 57 26 118
# верхние пределы ответа, воспроизводимого сервером из наблюдений сайта (защита от страниц с мегабайтами DATA)
reply_max_records 6
reply_max_bytes 8192
# каждая воспроизводимая запись ответа ограничена этим размером (у сайта DATA бывает 16401; полный размер = +16 КБ на соединение)
reply_rec_cap 2048

# --- форма первых записей самого туннеля (после обмена) ---
# клиент: первые up_records записей туннеля режутся на куски с телом up_chunk (QUIC Initial 1200+ Б не уходит одной записью;
# у браузера клиентские записи после HEADERS - десятки/сотни байт: 26, 30, 164)
up_chunk 150 470
up_records 10
# сервер: первые down_records записей туннеля добиваются набивкой до цели в [3/4*hi, hi], hi = min(down_pad_cap, самая крупная запись
# сайта после Finished); если сайт шлёт только мелкие записи (hi < 200) - набивки нет
down_records 4
down_pad_cap 1800
